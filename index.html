/** * US History EOC Study Guide: results logger * Bound to a Google Sheet (Extensions > Apps Script). Run setup() once, then deploy as a web app. * * Tabs it maintains: * Submissions every send, one row each (the raw log; never edited by the script after writing) * Students one row per student (latest send wins): domain % and standard % with red/green flags * Needs Help every domain and standard, with the students under the cutoff, ready for grouping */ const CUTOFF = 0.70; // under this share of attempted points = needs help (change if your team uses another line) const DOMAINS = [ ['D1', 'Colonization through the Constitution', ['SSUSH1','SSUSH2','SSUSH3','SSUSH4','SSUSH5']], ['D2', 'New Republic through Reconstruction', ['SSUSH6','SSUSH7','SSUSH8','SSUSH9','SSUSH10']], ['D3', 'Industrialization, Reform, & Imperialism', ['SSUSH11','SSUSH12','SSUSH13','SSUSH14']], ['D4', 'Establishment as a World Power', ['SSUSH15','SSUSH16','SSUSH17','SSUSH18','SSUSH19']], ['D5', 'Post-World War II to the Present', ['SSUSH20','SSUSH21','SSUSH22','SSUSH23']], ]; const STANDARDS = DOMAINS.flatMap(d => d[2]); const TOTAL_QUESTIONS = 102; const LOG_HEAD = ['Received', 'Last name', 'First name', 'Teacher', 'Block', 'Answered', 'Overall %', 'D1 %', 'D2 %', 'D3 %', 'D4 %', 'D5 %', 'Missed questions', 'Device', 'Raw']; const STU_HEAD = ['Last name', 'First name', 'Teacher', 'Block', 'Last sent', 'Times sent', 'Answered (of 102)', 'Overall %', 'D1 %', 'D2 %', 'D3 %', 'D4 %', 'D5 %', ...STANDARDS.map(s => s + ' %'), 'Missed questions', 'Key']; const HELP_HEAD = ['Domain or standard', 'Name', 'Students under ' + Math.round(CUTOFF * 100) + '%', 'Count', 'Not attempted']; // ---------- web app entry points ---------- function doGet(e) { const cb = String((e && e.parameter && e.parameter.cb) || '').replace(/[^\w$]/g, ''); const out = handle_((e && e.parameter && e.parameter.p) || ''); const text = cb ? cb + '(' + JSON.stringify(out) + ');' : JSON.stringify(out); return ContentService.createTextOutput(text) .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } function doPost(e) { const out = handle_((e && e.postData && e.postData.contents) || ''); return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } function handle_(raw) { try { const p = JSON.parse(raw); validate_(p); const lock = LockService.getScriptLock(); lock.waitLock(25000); try { record_(p, raw); } finally { lock.releaseLock(); } return { ok: true, received: new Date().toISOString() }; } catch (err) { return { ok: false, error: String((err && err.message) || err) }; } } function validate_(p) { if (!p || p.v !== 1) throw new Error('unrecognized data'); ['first', 'last', 'teacher', 'period'].forEach(k => { if (!p[k] || String(p[k]).trim() === '') throw new Error('missing ' + k); p[k] = String(p[k]).trim().slice(0, 60); }); if (!Array.isArray(p.dom) || p.dom.length !== 5) throw new Error('bad domain data'); if (typeof p.std !== 'object') throw new Error('bad standard data'); p.missed = String(p.missed || '').slice(0, 2000); } // ---------- writing ---------- function record_(p, raw) { const ss = SpreadsheetApp.getActiveSpreadsheet(); ensureSheets_(ss); const now = new Date(); const answered = p.dom.reduce((a, d) => a + (+d[2] || 0), 0); const earned = p.dom.reduce((a, d) => a + (+d[0] || 0), 0); const attempted = p.dom.reduce((a, d) => a + (+d[1] || 0), 0); const overall = attempted ? earned / attempted : ''; const domPct = p.dom.map(d => (+d[1] ? (+d[0]) / (+d[1]) : '')); const stdPct = STANDARDS.map(s => (p.std[s] && +p.std[s][1]) ? (+p.std[s][0]) / (+p.std[s][1]) : ''); ss.getSheetByName('Submissions').appendRow([now, p.last, p.first, p.teacher, p.period, answered, overall, ...domPct, p.missed, String(p.device || ''), String(raw).slice(0, 45000)]); const key = [p.last, p.first, p.teacher, p.period].map(x => String(x).toLowerCase().replace(/\s+/g, ' ')).join('|'); const stu = ss.getSheetByName('Students'); const keyCol = STU_HEAD.length; const last = stu.getLastRow(); let rowIdx = -1, times = 0; if (last > 1) { const keys = stu.getRange(2, keyCol, last - 1, 1).getValues(); for (let i = 0; i < keys.length; i++) if (keys[i][0] === key) { rowIdx = i + 2; break; } if (rowIdx > 0) times = +stu.getRange(rowIdx, 6).getValue() || 0; } const row = [p.last, p.first, p.teacher, p.period, now, times + 1, answered, overall, ...domPct, ...stdPct, p.missed, key]; if (rowIdx > 0) stu.getRange(rowIdx, 1, 1, row.length).setValues([row]); else stu.appendRow(row); rebuildNeedsHelp_(ss); } function rebuildNeedsHelp_(ss) { const stu = ss.getSheetByName('Students'); const help = ss.getSheetByName('Needs Help'); const n = stu.getLastRow() - 1; const data = n > 0 ? stu.getRange(2, 1, n, STU_HEAD.length).getValues() : []; const col = name => STU_HEAD.indexOf(name); const label = r => r[col('Last name')] + ', ' + r[col('First name')] + ' (' + r[col('Teacher')] + ' Block ' + r[col('Block')] + ')'; const rows = []; const add = (code, name, colName) => { const c = col(colName); const under = data.filter(r => r[c] !== '' && r[c] < CUTOFF).sort((a, b) => a[c] - b[c]); const none = data.filter(r => r[c] === '').length; rows.push([code, name, under.map(r => label(r) + ' ' + Math.round(r[c] * 100) + '%').join('\n'), under.length, none]); }; DOMAINS.forEach(d => { add(d[0], d[1], d[0] + ' %'); d[2].forEach(s => add(s, ' ' + d[0], s + ' %')); }); help.getRange(2, 1, Math.max(help.getLastRow() - 1, 1), HELP_HEAD.length).clearContent(); help.getRange(2, 1, rows.length, HELP_HEAD.length).setValues(rows); } // ---------- setup ---------- function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); ensureSheets_(ss); const stu = ss.getSheetByName('Students'); const pctCols = STU_HEAD.map((h, i) => h.endsWith('%') ? i + 1 : 0).filter(Boolean); const rules = []; pctCols.forEach(c => { const r = stu.getRange(2, c, 1000, 1); r.setNumberFormat('0%'); rules.push(SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied(`=AND(ISNUMBER(${colA1_(c)}2),${colA1_(c)}2<${CUTOFF})`) .setBackground('#F8D7D7').setFontColor('#8E2C2C').setRanges([r]).build()); rules.push(SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied(`=AND(ISNUMBER(${colA1_(c)}2),${colA1_(c)}2>=${CUTOFF})`) .setBackground('#DDF0E3').setFontColor('#2E7D4F').setRanges([r]).build()); }); stu.setConditionalFormatRules(rules); stu.hideColumns(STU_HEAD.length); const log = ss.getSheetByName('Submissions'); log.getRange(2, 7, 1000, 6).setNumberFormat('0%'); log.hideColumns(LOG_HEAD.length); const help = ss.getSheetByName('Needs Help'); help.setColumnWidth(1, 130); help.setColumnWidth(2, 300); help.setColumnWidth(3, 460); help.getRange(2, 3, 200, 1).setWrap(true).setVerticalAlignment('top'); rebuildNeedsHelp_(ss); } function ensureSheets_(ss) { [['Submissions', LOG_HEAD], ['Students', STU_HEAD], ['Needs Help', HELP_HEAD]].forEach(([name, head]) => { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0 || sh.getRange(1, 1).getValue() !== head[0]) { sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#1F3A5F').setFontColor('#FFFFFF'); sh.setFrozenRows(1); } }); } function colA1_(n) { let s = ''; while (n > 0) { const m = (n - 1) % 26; s = String.fromCharCode(65 + m) + s; n = Math.floor((n - 1) / 26); } return s; } /** Optional: run from the editor to confirm everything works before giving students the link. */ function testSubmission() { const p = { v: 1, first: 'Test', last: 'Student', teacher: 'Teacher', period: '1', device: 'test', dom: [[10, 12, 9, 17], [0, 0, 0, 21], [0, 0, 0, 16], [0, 0, 0, 24], [0, 0, 0, 24]], std: { SSUSH1: [3, 3], SSUSH2: [2, 4], SSUSH3: [5, 5] }, missed: 'Q4 2b, Q6 2d' }; Logger.log(JSON.stringify(handle_(JSON.stringify(p)))); }
